Aggregate labeling counts

Summarize overall, boxed-warning and hepatotoxicity labeling counts by year.

FDA_drug1 <- FDA_drug %>%
  dplyr::select(Year, Drug.Name, has_box_warning, box_warning_hepat_status, has_hepat) %>%
  dplyr::filter(Year <= 2025) %>%
  group_by(Year) %>%
  summarise(Total = n(),
            Box_Warning = sum(has_box_warning == "TRUE", na.rm = TRUE),
            Hepatotoxicity = sum(has_hepat == "TRUE", na.rm = TRUE),
            Box_Hepa = sum((box_warning_hepat_status == "TRUE"), na.rm = TRUE))

pie_data2 <- pie_data %>%
    group_by(source) %>%
    summarise(count = n()) %>%
    ungroup() %>%
    mutate(percentage = count / sum(count) * 100,
           label = paste0(source, "\n", round(percentage, 1), "%")) %>% 
    mutate(source = factor(source, levels = c("Animal study", "Clinical trial", "Postmarketing", "N/A")))

def run_enrichr(i):
    a1 = df.loc[i]
    a1 = list(a1[a1 < -5].index)

    if len(a1) <= 3:
        return f"{i}: too few genes ({len(a1)})"

    try:
        enr = gp.enrichr(
            gene_list=a1,
            gene_sets=gene_sets,
            organism='human',
            outdir=f'pathway/{i}',
            cutoff=1
        )

        if enr.results is None or enr.results.empty:
            return f"{i}: no enrichment results"

        if 'Adjusted P-value' not in enr.results.columns:
            return f"{i}: missing Adjusted P-value"

        return f"{i}: success ({len(enr.results)})"

    except Exception as e:
        return f"{i}: enrichr failed -> {e}"
